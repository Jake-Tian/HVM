# Requirements for preprocessing/whisperx_subtitles.py
# Generate subtitles using WhisperX

# Core dependencies
torch>=2.0.0,<2.6.0  # PyTorch (avoid 2.6+ due to weights_only breaking changes)
torchvision>=0.15.0,<0.19.0  # Must match PyTorch version for compatibility
torchaudio>=2.0.0,<2.6.0  # PyTorch audio processing
whisperx>=3.1.0  # WhisperX for transcription and diarization

# PyTorch 2.6 compatibility (if using PyTorch 2.6+, these help with omegaconf)
omegaconf>=2.3.0  # Configuration library used by pyannote

# Transformers (dependency of whisperx)
transformers>=4.30.0  # Hugging Face transformers

# Speaker diarization dependencies
pyannote.audio>=3.0.0  # Speaker diarization (requires HuggingFace token)
pyannote.core>=5.0.0

# Audio processing
faster-whisper>=0.9.0  # Faster Whisper backend

# Optional but recommended
numpy>=1.24.0
scipy>=1.10.0

# Installation notes:
# 1. Install Python packages:
#    pip install -r preprocessing/requirements_whisperx.txt
#
# 2. Install ffmpeg (system dependency, not a Python package):
#    Ubuntu/Debian: sudo apt-get install ffmpeg
#    macOS: brew install ffmpeg
#    Conda: conda install -c conda-forge ffmpeg
#
# 3. Set HuggingFace token for speaker diarization (optional but recommended):
#    export HF_TOKEN="your_huggingface_token_here"
#    Or the script will use a default token if available
#
# 4. For CUDA support, install PyTorch with CUDA:
#    pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118
